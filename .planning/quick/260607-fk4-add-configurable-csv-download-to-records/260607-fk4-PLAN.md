  <files>packages/web/src/lib/csvExport.ts, packages/web/src/lib/csvExport.spec.ts</files>
  <behavior>
    escapeCsvField(value: unknown): string
      - null/undefined → "" (empty, unquoted)
      - field containing comma, double-quote, CR, or LF → wrapped in double quotes; embedded `"` doubled (RFC 4180)
      - plain field (no special chars) → returned as-is, unquoted
      - non-string (number/boolean) → String()-coerced first, then escaped
    rowsToCsv(rows: Record<string, unknown>[], columns: string[]): string
      - first line = header row = columns joined (each header passed through escapeCsvField)
      - each data row emits ONLY the given columns, in the given order (extra row keys ignored; missing keys → "")
      - lines joined with "\r\n"; trailing newline acceptable but assert exact in tests
    sanitizeFilenamePart(raw: string): string
      - replaces characters illegal/awkward in filenames (/ \ : * ? " < > | and control chars and whitespace runs) with "-"
      - collapses repeated "-"; trims leading/trailing "-"; empty result → "table" (caller passes table when title empty)
    buildCsvFilename(titleOrTable: string, now: Date): string
      - returns `${sanitizeFilenamePart(titleOrTable)}-YYYYMMDD-HHmmss.csv` using LOCAL time, zero-padded
  </behavior>
  <action>

    - escapeCsvField: `const s = value == null ? "" : String(value); return /[",\r\n]/.test(s) ? '"' + s.replace(/"/g, '""') + '"' : s;`
