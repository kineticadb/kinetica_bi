      export function snapshotToLayerForm(snapshot: LayerSnapshot): { config: Record<string, unknown>; info: InfoPatch } {
        const config: Record<string, unknown> = {};

          if ((INFO_FIELDS as readonly string[]).includes(k)) {
            (info as Record<string, unknown>)[k] = v;            // info_* → separate props

          } else {

      export function layerFormToSnapshot(formConfig: Record<string, unknown>, infoPatch: InfoPatch = {}): LayerSnapshot {
