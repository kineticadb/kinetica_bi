```typescript
// packages/server/src/lib/aiConfig.ts — parse is PURE (env injected, never reads process.env, never exits, never fetches)
export type AiProvider = "openai-compatible" | "claude";
export type AiDataAccess = "schema" | "query";
export type AiConfig = {
  provider: AiProvider;
  baseUrl: string;            // claude default "https://api.anthropic.com/v1" when AI_BASE_URL unset
  model: string;
  dataAccess: AiDataAccess;   // default "schema"
  maxToolSteps: number;       // integer 1..50, default 8
  timeoutMs: number;          // integer 1000..600000, default 60000
  readonly apiKey?: string;   // NON-ENUMERABLE (Object.defineProperty, enumerable:false)
};
export type AiConfigResult =
  | { enabled: false; reason: "not_enabled" }
  | { enabled: false; reason: "invalid"; errors: string[] }
  | { enabled: true; config: AiConfig };
export function parseAiConfig(env: Record<string, string | undefined>): AiConfigResult;
export function formatAiConfigLog(result: AiConfigResult): string | null;
export function loadAiConfig(env?: Record<string, string | undefined>): AiConfigResult; // parse + log once + store
export function getAiConfig(): AiConfigResult; // last loaded result; { enabled:false, reason:"not_enabled" } before load
```
Exact error strings (names and reasons only — NEVER a value):
- `AI_ENABLED must be 'true' or 'false'`
- `AI_PROVIDER is missing`
- `AI_PROVIDER must be 'openai-compatible' or 'claude'` (for `anthropic` append ` (did you mean 'claude'?)`)
- `AI_MODEL is missing`
- `AI_BASE_URL is missing (required for AI_PROVIDER=openai-compatible)`
- `AI_BASE_URL is not a valid http(s) URL`
- `AI_API_KEY is missing (required for AI_PROVIDER=claude)`
- `AI_DATA_ACCESS must be 'schema' or 'query'`
- `AI_MAX_TOOL_STEPS must be an integer from 1 to 50`
- `AI_TIMEOUT_MS must be an integer from 1000 to 600000`
Log lines:
- not_enabled -> `null` (print nothing)
- invalid -> `[ai] AI is DISABLED — invalid configuration: <errors joined by "; ">. The app is running without AI.` (console.error)
- enabled -> `[ai] AI enabled: provider=<p> model=<m> endpoint=<new URL(baseUrl).origin> dataAccess=<d> maxToolSteps=<n> timeoutMs=<n> key=<set|not set> contract=<AI_CONTRACT_VERSION>` (console.log)
