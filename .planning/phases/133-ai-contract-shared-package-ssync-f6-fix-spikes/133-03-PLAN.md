NEW contract (packages/shared/src/aiOp.ts — later phases extend payloads deliberately):
```typescript
export const AI_OP_KINDS = ["patch","add_widget","remove_widget","add_layer","remove_layer",
  "add_dynamic_view","remove_dynamic_view","apply_filter","clear_filter"] as const;
export type AiOpKind = (typeof AI_OP_KINDS)[number];
export const AiTempIdSchema = z.number().int().negative();
export const AiRefIdSchema = z.number().int().refine((n) => n !== 0, { message: "id must be non-zero (positive = saved, negative = AI temp id)" });
export const AiActionTargetSchema = z.object({ kind: z.enum(TARGET_KINDS), id: AiRefIdSchema }).strict();
export const AiPatchActionSchema = z.object({ target: AiActionTargetSchema, configPatch: z.record(z.unknown()) }).strict();
export const AiOpSchema = z.discriminatedUnion("op", [
  z.object({ op: z.literal("patch"), action: AiPatchActionSchema }).strict(),
  z.object({ op: z.literal("add_widget"), tempId: AiTempIdSchema, type: z.string().min(1), title: z.string().max(200), config: z.record(z.unknown()) }).strict(),
  z.object({ op: z.literal("remove_widget"), id: AiRefIdSchema }).strict(),
  z.object({ op: z.literal("add_layer"), tempId: AiTempIdSchema, widgetId: AiRefIdSchema }).strict(),        // payload extended in Phase 136
  z.object({ op: z.literal("remove_layer"), id: AiRefIdSchema }).strict(),
  z.object({ op: z.literal("add_dynamic_view"), tempId: AiTempIdSchema }).strict(),                           // structured spec added in Phase 139
  z.object({ op: z.literal("remove_dynamic_view"), id: AiRefIdSchema }).strict(),
  z.object({ op: z.literal("apply_filter"), widgetId: AiRefIdSchema }).strict(),                              // filter payload added in Phase 138
  z.object({ op: z.literal("clear_filter"), widgetId: AiRefIdSchema.optional() }).strict(),
]);
export type AiOp = z.infer<typeof AiOpSchema>;
export function parseAiOp(raw: unknown): { ok: true; op: AiOp } | { ok: false; errors: string[] };
// errors = zod issues formatted as `${issue.path.join(".") || "(root)"}: ${issue.message}`
```
