```ts
export type AiAuditKind = "ask" | "generate" | "keep";
export type AiAuditDashboardState = "live" | "pending" | "deleted" | "none";
export type AiAuditRowDto = {
  id: number; ts: number; username: string;
  dashboardId: number | null; dashboardName: string | null; dashboardState: AiAuditDashboardState;
  kind: AiAuditKind;
  question: string;            // <= 161 chars ("…" appended when truncated); "" for keep rows
  summary: string | null;      // keep rows only, e.g. "Kept 3 added, 1 edited, 0 removed"
  outcome: string | null; outcomeDetail: string | null; steps: number | null; durationMs: number | null;
};
export type AiAuditListDto = { rows: AiAuditRowDto[]; total: number; page: number; pageSize: 50 };
export type AiAuditDetailDto = Omit<AiAuditRowDto, "question"> & {
  question: string; answerText: string | null;
  provider: string | null; model: string | null; dataAccess: string | null; allowListVersion: string | null;
  inputTokens: number | null; outputTokens: number | null;
  tools: { name: string; result: string; stage?: string; reasons?: string[] }[] | null;
  changes: { applied?: string[]; rejected?: { label: string; stage: string; reasons: string[] }[] } | { added?: unknown; edited?: unknown; removed?: unknown } | null;
  queries: { sql: string; rowCount?: number; durationMs?: number; outcome?: string; reason?: string }[] | null;
};
export type AiAuditDashboardsDto = { dashboards: { id: number; name: string | null; state: AiAuditDashboardState }[] };
```
- `GET /api/ai/audit?from=<ms>&to=<ms>&outcome=<csv>&dashboardId=<n>&page=<n>` -> AiAuditListDto (from/to default to the last 7 days server-side; outcome values: ok, cancelled, capped, provider_error, rate_limited, kept, keep_conflict, keep_denied, keep_invalid, keep_failed).
- `GET /api/ai/audit/:id` -> AiAuditDetailDto | 404 `{ error: "Not found." }`.
- `GET /api/ai/audit/dashboards?from=<ms>&to=<ms>` -> AiAuditDashboardsDto.
