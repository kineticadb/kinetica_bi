export type FormatSpec = FormatSpecNumber | FormatSpecDate | FormatSpecD3 | FormatSpecNone;

// Existing internal formatters all follow this never-throws → raw-value-fallback shape:
function buildD3Formatter(spec: FormatSpecD3): (v: unknown) => string | unknown {
  return (v: unknown): string | unknown => {
    if (v === null || v === undefined) return v;
    const n = typeof v === "number" ? v : Number(v);
    if (isNaN(n)) return v;
    try { return d3Format(spec.specifier)(n); } catch { return v; }
  };
}

// Existing buildFormatter switch (line 230-243) — exhaustiveness guard is the compile-time gate:
export function buildFormatter(spec: FormatSpec | null | undefined): (v: unknown) => string | unknown {
  if (spec == null || spec.kind === "none") return (v: unknown) => v;
  switch (spec.kind) {
    case "number": return buildNumberFormatter(spec);
    case "d3":     return buildD3Formatter(spec);
    case "date":   return buildDateFormatter(spec);
    default: {
      const _exhaustive: never = spec;  // <-- adding kind:"si" to the union without a case ERRORS here (intended)
      void _exhaustive;
      return (v: unknown) => v;
    }
  }
}

// defaultFormatKind returns FormatSpec["kind"] but MUST NEVER return "si" (SI is opt-in only — Pitfall 5).
