export const EXPORT_POLL_MS = 5000;
export type ExportTrackerState = { jobs: Record<string, ExportJobDto>; dialogJobId: string | null };
export const useExportTrackerStore: UseBoundStore<StoreApi<ExportTrackerState>>;

export function untrackExport(id: string): void;           // clear timer, drop entry, clear dialogJobId if it was this id
export function setDialogJob(id: string | null): void;     // null after a terminal job also drops that job's entry
export function stopAllExportTracking(): void;             // clear every timer, invalidate in-flight polls, reset store (logout)
