```typescript
export type FormatSpecNumber = {
  kind: "number";
  thousandsSep: boolean;     // true = grouping separator
  decimals: number;          // 0..N fixed decimal places
  currency: false | string;  // false = none; string = symbol PREFIX (default "$"), BEFORE the number
  percent: boolean;          // true = append LITERAL % WITHOUT ×100
};
export type FormatSpecDate = {
  kind: "date";
  preset: "iso" | "us" | "long" | "us_time" | "long_time" | "custom";
  customPattern?: string;    // only when preset === "custom"; tokens YYYY MM DD HH mm (ss optional)
};
export type FormatSpecD3 = {
  kind: "d3";
  specifier: string;         // raw d3 string; % DOES ×100 here (intentional escape hatch)
};
export type FormatSpecNone = { kind: "none" };
export type FormatSpec = FormatSpecNumber | FormatSpecDate | FormatSpecD3 | FormatSpecNone;

// buildFormatter(spec): (v: unknown) => string | unknown  — NEVER throws (raw fallback built in)
// defaultFormatKind(colName, columns): "number" | "date" | "d3" | "none"
//   (number→"number", datetime→"date", else→"none")
```

us_time=`06/19/2026 HH:mm`, long_time=`Jun 19, 2026 HH:mm`, plus `custom` (token pattern).
Surface EXACTLY these 6 options in the date preset dropdown (custom shows the pattern input).
