Make a single Radio Dashboard Control OPTION drive MULTIPLE targets at once — the model + dispatch + wiring layer (RADIOMULTI-V111-01).

`RadioOption.action: WidgetAction` becomes `actions: WidgetAction[]` (ordered, mixed widget/layer/dynamicView kinds). A back-compat normalizer `getOptionActions(option)` reads either the new `actions` array OR a legacy persisted single `action` (NO DB migration). A new `applyWidgetActions(actions, lookups, controlId)` assembles ONE combined contribution from all valid actions and calls `setControlContribution` EXACTLY ONCE — giving OPTION-LEVEL switch-replace for free (switching options drops targets the new option doesn't set). The renderer + context + production dispatch closure are wired to the plural path.

Purpose: This is the correctness-critical core of the phase. The overlay store is ALREADY control-keyed/multi-target — no store change. The bug to avoid: do NOT loop the per-target-merge `applyWidgetAction` (that strands stale targets).

    ```typescript
    export function applyWidgetActions(
      actions: WidgetAction[],
      lookups: ActionLookups,
      controlId: number,
    ): WidgetActionsResult {
      const combined: ControlContribution = { widget: {}, layer: {}, dynamicView: {} };
      const applied: WidgetActionTarget[] = [];
      const rejected: WidgetActionsResult["rejected"] = [];
      const notFound: WidgetActionTarget[] = [];

      for (const { target, configPatch } of actions) {
        // --- resolve target (mirror singular Step 1) ---
        let widgetType: string | undefined;
        if (target.kind === "widget") {
          const w = lookups.widgets.find((x) => x.id === target.id);
          if (!w) { notFound.push(target); continue; }
          widgetType = w.type;
        } else if (target.kind === "layer") {
          if (!lookups.layers.find((x) => x.id === target.id)) { notFound.push(target); continue; }
        } else if (!lookups.dynamicViewIds.includes(target.id)) { notFound.push(target); continue; }

        // --- validate (mirror singular Step 2: layer → snapshot denylist; else strict allow-list) ---
        const validation = target.kind === "layer"
          ? validateLayerSnapshot(configPatch)
          : validateActionPatch(target.kind, widgetType, configPatch);
        if (!validation.valid) { rejected.push({ target, reasons: validation.reasons }); continue; }

        // --- build the per-target patch (mirror singular Step 3) + write into the COMBINED contribution ---
        if (target.kind === "layer") {
          const split = splitLayerSnapshot(configPatch);
          combined.layer[target.id] = { ...split.topLevel, ...(split.config ? { config: split.config } : {}) };
        } else if (target.kind === "widget") {
          combined.widget[target.id] = configPatch;
        } else {
          combined.dynamicView[target.id] = configPatch;
        }
        applied.push(target);
      }

      // --- best-effort failure surfacing: ONE combined toast (mirror singular toast) ---
      if (rejected.length > 0 || notFound.length > 0) {
        const parts: string[] = [];
        if (rejected.length > 0) parts.push(`${rejected.length} rejected: ${rejected.flatMap(r => r.reasons).join("; ")}`);
        if (notFound.length > 0) parts.push(`${notFound.length} target(s) no longer available`);
        useToastStore.getState().showToast(`Some option targets were skipped — ${parts.join(" · ")}`, "info");
      }

      // --- ONE wholesale write (switch-replace at the option level). Skip entirely if nothing valid. ---
      if (applied.length === 0) {
        // No valid actions: still REPLACE the control's contribution with an empty one so a prior
        // option's targets are dropped (switch-replace must hold even when the new option is all-invalid).
        // BUT only if the control HAD a prior contribution — avoid a spurious write/derive otherwise.
        const had = controlId in useWidgetActionStore.getState().contributions;
        if (had) useWidgetActionStore.getState().setControlContribution(controlId, combined);
        return { status: "noop", applied, rejected, notFound };
      }

      useWidgetActionStore.getState().setControlContribution(controlId, combined);
      return {
        status: rejected.length > 0 || notFound.length > 0 ? "partial" : "applied",
        applied, rejected, notFound,
      };
    }
    ```
