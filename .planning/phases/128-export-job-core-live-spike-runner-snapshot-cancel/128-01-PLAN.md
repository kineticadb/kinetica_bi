export function escapeCsvField(value: unknown): string {
  const s = value == null ? "" : String(value);
  return /[",\r\n]/.test(s) ? '"' + s.replace(/"/g, '""') + '"' : s;
}
export function rowsToCsv(rows: Record<string, unknown>[], columns: string[]): string {
  const lines: string[] = [columns.map(escapeCsvField).join(",")];
  for (const row of rows) lines.push(columns.map((c) => escapeCsvField(row[c])).join(","));
  return lines.join("\r\n");   // CRLF BETWEEN lines, no trailing CRLF, no BOM
}
