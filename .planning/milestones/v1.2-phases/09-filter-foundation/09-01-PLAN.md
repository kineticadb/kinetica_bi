```typescript
import { create } from "zustand";
import { useToastStore } from "./toast";

// PITFALL C-04 / AP-4 lock: filters MUST be Record<tableId, ActiveFilter[]> from day 1.
// Retrofit cost is very high — see .planning/research/PITFALLS.md C-04.
//
// distinction note: this store is the CLIENT-TRANSIENT drill-down filter set.
// The existing `views.filter_clause` field on the server (see ViewDto in src/api/client.ts)
// is the SERVER-PERSISTED filter applied at view-materialize time — totally separate concept.
// Phase 10 surfaces both in one filter bar; Phase 9 just builds this client-side store.

const FILTER_CAP_PER_TABLE = 10; // PITFALL D-04 lock — never silently drop; warn user

export type ActiveFilter = {
  column: string;
  value: string | number | boolean | Date | null;
  dataType: "string" | "number" | "boolean" | "datetime" | "null";
  sourceWidgetId?: number; // optional — Phase 10's DRILL-04 highlight uses it; tests skip it
  addedAt: number;         // Date.now() at addFilter time
};

export type FilterState = {
  filters: Record<number, ActiveFilter[]>;
  filterVersion: number;
  // Open Question #1 resolution (RESEARCH.md): tableId is a SEPARATE param, not on ActiveFilter.
  addFilter: (tableId: number, filter: ActiveFilter) => void;
  removeFilter: (tableId: number, column: string) => void;
  clearFilters: (tableId: number) => void;
  // reset() is internal: called from App.tsx on logout, DashboardsPage on dashboard switch.
  reset: () => void;
};

export const useFilterStore = create<FilterState>((set) => ({
  filters: {},
  filterVersion: 0,

  addFilter: (tableId, filter) =>
    set((state) => {
      const existing = state.filters[tableId] ?? [];

      // PITFALL D-04: 10-cap. If we'd exceed, no-op + toast.
      // Note: REPLACE on same column is allowed even at cap (count doesn't increase).
      const sameColumnIdx = existing.findIndex((f) => f.column === filter.column);
      if (sameColumnIdx === -1 && existing.length >= FILTER_CAP_PER_TABLE) {
        useToastStore.getState().showToast(
          "Filter limit reached (10 per table). Clear some first.",
          "info"
        );
        return state; // no state change, no version bump
      }

      // Exact-duplicate dedupe (same column AND same value) — silent no-op (matches "click selected = stay selected" UX)
      if (
        sameColumnIdx !== -1 &&
        existing[sameColumnIdx].value === filter.value
      ) {
        return state; // silent — no toast, no version bump
      }

      // PITFALL D-05 lock: same-column-different-value REPLACES, never appends.
      // Single transactional set() — no intermediate state where col=A AND col=B can be observed.
      let newTableFilters: ActiveFilter[];
      if (sameColumnIdx !== -1) {
        newTableFilters = [...existing];
        newTableFilters[sameColumnIdx] = filter;
      } else {
        newTableFilters = [...existing, filter];
      }

      return {
        filters: { ...state.filters, [tableId]: newTableFilters },
        filterVersion: state.filterVersion + 1, // PITFALL S-02 lock: primitive dep MUST advance
      };
    }),

  removeFilter: (tableId, column) =>
    set((state) => {
      const existing = state.filters[tableId] ?? [];
      const next = existing.filter((f) => f.column !== column);
      if (next.length === existing.length) return state; // nothing removed — no version bump
      return {
        filters: { ...state.filters, [tableId]: next },
        filterVersion: state.filterVersion + 1,
      };
    }),

  clearFilters: (tableId) =>
    set((state) => {
      const existing = state.filters[tableId] ?? [];
      if (existing.length === 0) return state; // nothing to clear — no version bump
      const next = { ...state.filters };
      delete next[tableId]; // PITFALL S-02: deletion semantics — empty key vs absent key both selector-safe
      return {
        filters: next,
        filterVersion: state.filterVersion + 1,
      };
    }),

  // Internal-only — called from App.tsx on logout and DashboardsPage on dashboard switch.
  // NOT exposed as a user action.
  reset: () => set({ filters: {}, filterVersion: 0 }),
}));

/* ------------------------------------------------------------------ */
/*  SQL-safe filter builder utilities                                  */
/* ------------------------------------------------------------------ */
//
// PITFALL D-02 / AP-3 lock: escapeKineticaStringLiteral is the ONE sanctioned
// path for interpolating user-clicked string values into SQL. NO EXCEPTIONS.
// Replaces single-quote with double-single-quote (SQL standard escape).

export const escapeKineticaStringLiteral = (s: string): string => {
  return s.replace(/'/g, "''");
};

// PITFALL D-03 lock: never `column = NULL` (SQL three-valued logic makes it always-false).
// All null values flow through the IS NULL branch.
//
// NOTE on TIMESTAMP literal format (RESEARCH.md Open Question #3 — LOW confidence):
// Phase 9 has no UI surface that produces datetime clicks. The branch below uses the
// most-likely Kinetica syntax `TIMESTAMP 'YYYY-MM-DD HH:MM:SS'`. Real validation happens
// when Phase 10 adds drill-down click paths that produce datetime filters. If the format
// turns out to be wrong, the fix is one branch in this function. Phase 9 is NOT blocked.
export const buildEqualityFilter = (filter: ActiveFilter): string => {
  if (filter.dataType === "null" || filter.value === null) {
    return `${filter.column} IS NULL`;
  }
  if (filter.dataType === "string") {
    const escaped = escapeKineticaStringLiteral(String(filter.value));
    return `${filter.column} = '${escaped}'`;
  }
  if (filter.dataType === "number") {
    return `${filter.column} = ${filter.value}`;
  }
  if (filter.dataType === "boolean") {
    return `${filter.column} = ${filter.value ? "TRUE" : "FALSE"}`;
  }
  if (filter.dataType === "datetime") {
    // Format: ISO-like SQL TIMESTAMP literal. UNTESTED against deployed Kinetica in Phase 9
    // (no Phase 9 path generates datetime clicks). Phase 10 click handlers validate this.
    const d = filter.value instanceof Date
      ? filter.value
      : new Date(filter.value as string | number);
    const iso = d.toISOString().replace("T", " ").replace(/\.\d+Z$/, "");
    return `${filter.column} = TIMESTAMP '${iso}'`;
  }
  // Exhaustiveness — should be unreachable given the dataType union
  throw new Error(`Unhandled ActiveFilter.dataType: ${(filter as { dataType: string }).dataType}`);
};

export const buildWhereClause = (filters: ActiveFilter[]): string => {
  if (filters.length === 0) return "";
  return filters.map(buildEqualityFilter).join(" AND ");
};

// Splices a WHERE clause into baseSql.
// Strategy: regex/string-scan for GROUP BY | ORDER BY | LIMIT — full SQL parsing is overkill
// for the predictable two-pattern SQL shapes that ChartConfigPanel.tsx generates (verified
// at .planning/phases/09-filter-foundation/09-RESEARCH.md "SQL Assembly in ChartConfigPanel").
//
// Cases handled:
//   1. empty whereClause → return baseSql unchanged
//   2. no existing WHERE, has GROUP BY/ORDER BY/LIMIT → insert WHERE before earliest of those keywords
//   3. no existing WHERE, no trailing keywords → append WHERE at end
//   4. existing WHERE → splice ` AND <new>` before earliest of GROUP BY/ORDER BY/LIMIT (or append at end)
//   5. defensive: strip leading "WHERE " from whereClause input
export const injectWhereClause = (baseSql: string, whereClause: string): string => {
  if (!whereClause || !whereClause.trim()) return baseSql;
  // Strip leading "WHERE " if caller accidentally included it
  const clause = whereClause.replace(/^\s*WHERE\s+/i, "");

  // Find the earliest position of GROUP BY / ORDER BY / LIMIT (case-insensitive, word-boundary).
  // Use regex with capture index to get the position. Match on whole word to avoid colliding
  // with column names that happen to contain those substrings (defensive).
  const trailingKeywordRegex = /\s+(GROUP\s+BY|ORDER\s+BY|LIMIT)\b/i;
  const match = trailingKeywordRegex.exec(baseSql);
  const trailingIdx = match ? match.index : -1;

  // Detect existing WHERE (word-boundary, case-insensitive)
  const whereMatch = /\bWHERE\b/i.exec(baseSql);

  if (whereMatch) {
    // Existing WHERE — append " AND <clause>" before trailing keyword (or at end)
    if (trailingIdx === -1) {
      return `${baseSql} AND ${clause}`;
    }
    return `${baseSql.slice(0, trailingIdx)} AND ${clause}${baseSql.slice(trailingIdx)}`;
  }

  // No existing WHERE — insert " WHERE <clause>" before trailing keyword (or at end)
  if (trailingIdx === -1) {
    return `${baseSql} WHERE ${clause}`;
  }
  return `${baseSql.slice(0, trailingIdx)} WHERE ${clause}${baseSql.slice(trailingIdx)}`;
};
```
