const AggregatedWidgetRenderer = ({ widget }: Props) => {
  const [data, setData] = useState<Row[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const cfg = widget.config ?? {};
  const sql = cfg.sql as string | undefined;
  const tableId = cfg.tableId as number | undefined; // Plan 09-02 saves this; may be undefined for legacy widgets

  // PITFALL C-02 lock: scope selector to filters[tableId] — NOT state.filters whole.
  // PITFALL S-02 lock: filterVersion is the primitive dep for useEffect — never the array itself.
  const tableFilters = useFilterStore((state) => (tableId !== undefined ? state.filters[tableId] ?? [] : []));
  const filterVersion = useFilterStore((state) => state.filterVersion);

  useEffect(() => {
    if (!sql?.trim()) {
      setData([]);
      return;
    }
    const controller = new AbortController(); // Phase 9 FILT-02: cancel in-flight on filter change
    const whereClause = buildWhereClause(tableFilters);
    const finalSql = injectWhereClause(sql, whereClause);
    setLoading(true);
    setError(null);
