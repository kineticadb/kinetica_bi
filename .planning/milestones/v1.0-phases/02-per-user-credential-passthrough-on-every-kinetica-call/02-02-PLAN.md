    import { randomUUID } from "node:crypto"; // for fallback if req.requestId missing
    import type { AuthedRequest } from "./auth";
    import {
      KineticaAuthError,
      KineticaPermissionError,
      KineticaUpstreamError,
    } from "./kineticaErrors";

    export type KineticaOp = "SQL" | "DISCOVERY" | "MATERIALIZE" | "WMS";

    export type KineticaSqlOptions = {
      route: string;
      op: KineticaOp; // narrowing: routes pass SQL | DISCOVERY | MATERIALIZE
      extra?: Record<string, unknown>;
    };

    export type KineticaWmsOptions = { route: string };

    type AuditOutcome = "success" | "auth-fail" | "permission-denied" | "upstream-error";

    const buildAuthHeader = (creds: { username: string; password: string }): string =>
      `Basic ${Buffer.from(`${creds.username}:${creds.password}`).toString("base64")}`;

    const emitAudit = (fields: {
      request_id: string;
      username: string;
      route: string;
      op: KineticaOp;
      outcome: AuditOutcome;
      status: number;
      duration_ms: number;
    }): void => {
      console.log(
        JSON.stringify({
          ts: new Date().toISOString(),
          ...fields,
        })
      );
    };

    // Outcome classifier for HTTP status. body.status==='ERROR' on a 200 → upstream-error.
    const classifyAndThrow = (status: number, message: string): never => {
      if (status === 401) throw new KineticaAuthError(message, 401);
      if (status === 403) throw new KineticaPermissionError(message, 403);
      throw new KineticaUpstreamError(message, status);
    };

    export const kineticaSql = async (
      req: AuthedRequest,
      sql: string,
      options: KineticaSqlOptions
    ): Promise<unknown> => {
      const start = Date.now();
      const username = req.user?.creds?.username ?? "unknown";
      const requestId = req.requestId ?? randomUUID();
