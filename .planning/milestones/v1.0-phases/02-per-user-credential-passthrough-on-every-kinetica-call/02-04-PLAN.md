Existing GET /api/wms at index.ts:413-439:
```typescript
app.get("/api/wms", requireConfig, async (req, res) => {
  const kineticaUrl = process.env.KINETICA_URL;
  const baseUrl = kineticaUrl!.replace(/\/$/, "");
  const queryString = new URLSearchParams(req.query as Record<string, string>).toString();
  const wmsUrl = `${baseUrl}/wms?${queryString}`;
  try {
    const response = await fetch(wmsUrl, {
      headers: { Authorization: `Basic ${Buffer.from(`${kineticaUser}:${kineticaPassword}`).toString("base64")}` }
    });
    if (!response.ok) return res.status(response.status).send(await response.text());
    const contentType = response.headers.get("content-type");
    if (contentType) res.setHeader("Content-Type", contentType);
    const buffer = Buffer.from(await response.arrayBuffer());
    return res.send(buffer);
  } catch (error) {
    console.error("WMS proxy error", error);
    return res.status(502).json({ error: "Failed to reach Kinetica WMS", detail: String(error) });
  }
});
```

After refactor: kineticaWms throws on 401/403/non-ok; route only handles successful Response (or catches typed errors). The `console.error` is deleted (helper owns logging).
