```typescript
import { useFilterStore } from "../store/filterStore";
import { useSpatialFilterStore } from "../store/spatialFilterStore";

// Phase 109 (FCLEAR-V120-01): global clear-all. INPUT-STORE MUTATIONS ONLY.
// Loops the existing per-source clear actions over every active source, then clears
// spatial shapes. Never calls materialize/drop and never filterStore.reset() live —
// the untouched useCombinationOrchestrator ref-count DROPs the now-unused views.
// No-op safe: empty stores => zero iterations + clearAll() self-no-ops.
export function clearAllFilters(): void {
  const fs = useFilterStore.getState();
  for (const tableId of Object.keys(fs.filters)) {
    fs.clearFilters(Number(tableId));
  }
  for (const dvId of Object.keys(fs.dvFilters)) {
    fs.clearDvFilters(Number(dvId));
  }
  useSpatialFilterStore.getState().clearAll();
}
```

Capture `getState()` snapshots for the key lists BEFORE looping (Object.keys over the
snapshot) so the functional `set()` deletes don't disturb iteration.
